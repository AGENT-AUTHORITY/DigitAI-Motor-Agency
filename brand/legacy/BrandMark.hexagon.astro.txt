---
/**
 * BrandMark — isotipo de digitAI Motor.
 *
 * CONCEPTO
 * Hexagono exterior = el sistema completo: seis caras para los seis nodos que
 * la marca nombra (traffic, creative, landing, lead, sales, revenue).
 * Retícula interior rotada 30 grados = la estructura que lo sostiene, visible
 * solo de cerca. Dentro, una M: la inicial de Motor y, leida como grafico, una
 * onda de senal. Su valle central es el punto bajo, y ese punto lleva el nodo
 * de acento porque es exactamente lo que la marca promete encontrar. El nodo
 * verde corona la subida siguiente: el resultado.
 *
 * GEOMETRIA
 * Hexagono pointy-top de circunradio 20 sobre centro (24,24). La retícula es
 * el mismo hexagono girado 30 grados a radio 11, la relacion sqrt(3)/2 entre
 * hexagonos inscritos. La M es simetrica respecto del eje vertical: sus dos
 * astas caen en x=12 y x=36, equidistantes del centro, y el valle en el eje.
 *
 * El verde es semantico, igual que en el resto del sistema: marca resultado,
 * nunca decora.
 *
 * TAMANOS CHICOS
 * `detail={false}` quita la retícula interior. Debajo de ~24px esa capa solo
 * ensucia, y el hexagono con el vector sigue leyendose. El favicon usa esa
 * variante.
 */
export interface Props {
  size?: number;
  /** Retícula interior. Apagar bajo ~24px. */
  detail?: boolean;
  /** Todo en currentColor, para una tinta / impresion / fondos raros. */
  mono?: boolean;
  /** Animacion de recorrido del vector al entrar. */
  animated?: boolean;
  class?: string;
  title?: string;
}

const {
  size = 32,
  detail = true,
  mono = false,
  animated = false,
  class: className = '',
  title,
} = Astro.props;

// Hexagono exterior, pointy-top, R = 20
const OUTER = 'M24 4 L41.32 14 L41.32 34 L24 44 L6.68 34 L6.68 14 Z';
// Retícula interior, rotada 30 grados, R = 11.5
const INNER = 'M35 24 L29.5 14.47 L18.5 14.47 L13 24 L18.5 33.53 L29.5 33.53 Z';
// Recorrido: vertice inferior izquierdo -> centro -> vertice superior
const PATH = 'M12 34 L12 16 L24 28 L36 16 L36 34';
---

<svg
  class:list={['brandmark', mono && 'brandmark--mono', animated && 'brandmark--animated', className]}
  width={size}
  height={size}
  viewBox="0 0 48 48"
  fill="none"
  role={title ? 'img' : 'presentation'}
  aria-label={title}
  aria-hidden={title ? undefined : 'true'}
  focusable="false"
>
  {title && <title>{title}</title>}

  {detail && <path class="bm-lattice" d={INNER} />}

  <path class="bm-frame" d={OUTER} />

  <path class="bm-path" d={PATH} />

  <circle class="bm-node bm-node--origin" cx="12" cy="34" r="2.4" />
  <circle class="bm-node bm-node--core" cx="24" cy="28" r="3.5" />
  <circle class="bm-node bm-node--result" cx="36" cy="16" r="3.1" />
</svg>
